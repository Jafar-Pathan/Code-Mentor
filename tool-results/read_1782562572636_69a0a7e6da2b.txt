    91→// ── Code block with copy button ────────────────────────────────────────
    92→function CodeBlock({ children, className }: { children: string; className?: string }) {
    93→  const [copied, setCopied] = useState(false);
    94→  const handleCopy = useCallback(() => {
    95→    navigator.clipboard.writeText(children.trim());
    96→    setCopied(true);
    97→    setTimeout(() => setCopied(false), 2000);
    98→  }, [children]);
    99→
   100→  const lang = className?.replace('language-', '') || 'code';
   101→
   102→  return (
   103→    <div className="relative group my-3">
   104→      <div className="flex items-center justify-between px-4 py-2 bg-black/40 border border-border rounded-t-lg">
   105→        <span className="text-xs text-muted-foreground font-mono">{lang}</span>
   106→        <Button
   107→          variant="ghost"
   108→          size="sm"
   109→          className="h-6 px-2 text-xs text-muted-foreground hover:text-foreground opacity-0 group-hover:opacity-100 transition-opacity"
   110→          onClick={handleCopy}
   111→        >
   112→          {copied ? <Check className="w-3 h-3 text-success mr-1" /> : <Copy className="w-3 h-3 mr-1" />}
   113→          {copied ? 'Copied' : 'Copy'}
   114→        </Button>
   115→      </div>
   116→      <div className="code-block rounded-t-none !rounded-t-none">
   117→        <pre className="p-4 overflow-x-auto">
   118→          <code className={className}>{children}</code>
   119→        </pre>
   120→      </div>
   121→    </div>
   122→  );
   123→}
