   113→// ── Code block with copy button ────────────────────────────────────────
   114→function CodeBlock({ children, className }: { children: string; className?: string }) {
   115→  const [copied, setCopied] = useState(false);
   116→  const handleCopy = useCallback(() => {
   117→    navigator.clipboard.writeText(children.trim());
   118→    setCopied(true);
   119→    setTimeout(() => setCopied(false), 2000);
   120→  }, [children]);
   121→
   122→  const lang = className?.replace('language-', '') || 'code';
   123→
   124→  return (
   125→    <div className="relative group my-3">
   126→      <div className="flex items-center justify-between px-4 py-2 bg-black/40 border border-border rounded-t-lg">
   127→        <span className="text-xs text-muted-foreground font-mono">{lang}</span>
   128→        <Button
   129→          variant="ghost"
   130→          size="sm"
   131→          className="h-6 px-2 text-xs text-muted-foreground hover:text-foreground opacity-0 group-hover:opacity-100 transition-opacity"
   132→          onClick={handleCopy}
   133→        >
   134→          {copied ? <Check className="w-3 h-3 text-success mr-1" /> : <Copy className="w-3 h-3 mr-1" />}
   135→          {copied ? 'Copied' : 'Copy'}
   136→        </Button>
   137→      </div>
   138→      <div className="code-block rounded-t-none !rounded-t-none">
   139→        <pre className="p-4 overflow-x-auto">
   140→          <code className={className}>{children}</code>
   141→        </pre>
   142→      </div>
   143→    </div>
   144→  );
   145→}
